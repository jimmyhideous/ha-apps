#include <tunables/global>

# Parent: s6-overlay and the bashio scripts. Child profiles: alloy, nginx.
# The Supervisor loads this as "<repository hash>_ha_alloy"; peers match "*_ha_alloy".

profile ha_alloy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  file,
  signal (send) set=(kill,term,int,hup,cont),
  signal (receive) peer=unconfined,

  # s6-overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,
  /dev/tty rw,

  # bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,
  /data/** rw,

  /usr/bin/alloy cx -> alloy,
  /usr/sbin/nginx cx -> nginx,

  profile alloy flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    signal (receive) peer=*_ha_alloy,
    signal (send, receive) peer=*_ha_alloy//alloy,

    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,

    /usr/bin/alloy rm,

    # Built-in pipeline, generated config, user folder
    /etc/alloy/{,**} r,
    /run/alloy/{,**} r,
    /config/{,**} r,

    /data/ r,
    /data/alloy/ rw,
    /data/alloy/** rwk,

    /var/log/journal/{,**} r,
    /run/log/journal/{,**} r,
    /etc/machine-id r,
    /var/lib/dbus/machine-id r,
    @{PROC}/sys/kernel/random/boot_id r,

    /etc/{hosts,host.conf,resolv.conf,nsswitch.conf,gai.conf,services} r,
    /etc/{passwd,group} r,
    /etc/ssl/certs/{,**} r,
    /usr/share/ca-certificates/{,**} r,
    /usr/share/zoneinfo/{,**} r,

    # Go runtime
    @{PROC}/@{pid}/{,**} r,
    @{PROC}/stat r,
    @{PROC}/sys/net/core/somaxconn r,
    @{sys}/kernel/mm/transparent_hugepage/hpage_pmd_size r,
    @{sys}/fs/cgroup/{,**} r,
    @{sys}/devices/system/cpu/{,**} r,

    /dev/tty rw,
  }

  profile nginx flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Workers run as www-data.
    capability setuid,
    capability setgid,
    capability chown,

    signal (receive) peer=*_ha_alloy,
    signal (send, receive) peer=*_ha_alloy//nginx,

    network inet stream,
    network inet6 stream,

    /usr/sbin/nginx rm,
    /etc/nginx/{,**} r,
    /run/nginx/{,**} rw,
    /etc/{passwd,group,nsswitch.conf} r,
    @{PROC}/@{pid}/{,**} r,
    @{sys}/devices/system/cpu/{,**} r,

    /dev/tty rw,
  }
}
