#include <tunables/global>

# AppArmor profile for the HA Alloy app.
#
# Two layers, as in the Home Assistant app template (home-assistant/apps-example):
# - The parent profile covers s6-overlay (PID 1, s6-rc, s6-supervise) and the bashio run/finish scripts. The
#   Supervisor renames it to "<repository hash>_ha_alloy" when it loads it, so peers below use
#   "*_ha_alloy". It keeps the template's broad file rule, but has NO network rules, so nothing except
#   Alloy can open a socket.
# - Grafana Alloy, the only long-running process (it parses journal lines and talks to Loki), switches into the
#   child profile "alloy", which lists every file it may touch.
#
# To debug a denial: add "complain" to the flags of the affected profile, run the app, read
# `journalctl _TRANSPORT="audit" -g 'apparmor="ALLOWED"'` on the host, add the missing rule, remove "complain".

profile ha_alloy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # s6-overlay and bashio need wide file access while the container boots (template default).
  file,
  # s6 stops and restarts its services with these signals.
  signal (send) set=(kill,term,int,hup,cont),
  # Docker/the Supervisor send SIGTERM to PID 1 from outside the profile on stop.
  signal (receive) peer=unconfined,

  # S6-Overlay: init, its binaries, the compiled service tree in /run and the service definitions.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio (the run and finish scripts), and its temp files.
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # options.json (read by bashio::config) and the app's persistent data.
  /data/** rw,

  # Start Grafana Alloy in its own, tighter profile.
  /usr/bin/alloy cx -> alloy,

  profile alloy flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Stop/restart signals from s6-supervise (parent profile).
    signal (receive) peer=*_ha_alloy,
    # The Go runtime signals its own threads (SIGURG for goroutine preemption).
    signal (send, receive) peer=*_ha_alloy//alloy,

    # Network: HTTP(S) pushes to Loki and DNS lookups over IPv4/IPv6 (TCP and UDP), plus Alloy's own HTTP
    # server on 127.0.0.1:12345. netlink raw lets Go list network interfaces/addresses (read-only route info).
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,

    # The Alloy binary itself (read + map for execution).
    /usr/bin/alloy rm,

    # The pipeline configuration, read-only.
    /etc/alloy/{,**} r,

    # --storage.path=/data/alloy: journal read position and component state (read/write, file locks).
    # The rest of /data (options.json) is not needed by Alloy.
    /data/ r,
    /data/alloy/ rw,
    /data/alloy/** rwk,

    # The host journal, read-only: persistent (/var/log/journal, mounted by `journald: true`) and volatile
    # (/run/log/journal).
    /var/log/journal/{,**} r,
    /run/log/journal/{,**} r,

    # IDs that libsystemd's journal reader may look up. The machine-id files usually do not exist in the
    # container; allowing the read makes the lookup fail with ENOENT instead of a permission error.
    /etc/machine-id r,
    /var/lib/dbus/machine-id r,
    @{PROC}/sys/kernel/random/boot_id r,

    # Name resolution (Go resolver) and user lookups.
    /etc/{hosts,host.conf,resolv.conf,nsswitch.conf,gai.conf,services} r,
    /etc/{passwd,group} r,

    # CA certificates for HTTPS to Loki, and time zone data.
    /etc/ssl/certs/{,**} r,
    /usr/share/ca-certificates/{,**} r,
    /usr/share/zoneinfo/{,**} r,

    # Go runtime and Alloy's process metrics: its own /proc entries, boot time, listen backlog, huge page size,
    # and the cgroup limits/CPU topology used to size GOMAXPROCS.
    @{PROC}/@{pid}/{,**} r,
    @{PROC}/stat r,
    @{PROC}/sys/net/core/somaxconn r,
    @{sys}/kernel/mm/transparent_hugepage/hpage_pmd_size r,
    @{sys}/fs/cgroup/{,**} r,
    @{sys}/devices/system/cpu/{,**} r,

    # Log output inherited from s6 (as in the template's child profile).
    /dev/tty rw,
  }
}
